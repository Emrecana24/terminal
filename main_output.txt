    if (reqUrl.pathname === '/api/exec' && req.method === 'POST') {

        req.on('end', () => {
            try {
                const { command } = JSON.parse(body);
                exec(command, { cwd: ROOT_DIR, maxBuffer: 1024 * 1024 * 5 }, (error, stdout, stderr) => {
                    res.writeHead(200, { 'Content-Type': 'application/json; charset=utf-8' });
                    res.end(JSON.stringify({

                        stderr: stderr || '',
                        exitCode: error ? error.code : 0,
                        error: error ? error.message : null
                    }));

            }
        });

                } catch(err) {}
                return {
                    name: e.name,
                    isDir: e.isDirectory(),
                    size: size,
                    fullPath: path.join(targetDir, e.name)
                };
            });
            list.sort((a, b) => (a.isDir === b.isDir) ? a.name.localeCompare(b.name) : (a.isDir ? -1 : 1));
            res.writeHead(200, { 'Content-Type': 'application/json; charset=utf-8' });
            res.end(JSON.stringify({ currentDir: targetDir, parentDir: path.dirname(targetDir), items: list }));
        } catch (err) {

            res.end(JSON.stringify({ error: err.message }));

    }

        try {

            res.writeHead(200, { 'Content-Type': 'application/json; charset=utf-8' });
            res.end(JSON.stringify({ content, name: path.basename(filePath) }));
        } catch (err) {

          const functionCall = candidate.parts?.find(p => p.functionCall)?.functionCall;

                }

          }

    const term = new Terminal({
      cursorBlink: true,
      fontFamily: '"DejaVu Sans Mono", monospace',
      fontSize: userFontSize,
      lineHeight: 1.1,
      theme: { background: '#0d1117', foreground: '#c9d1d9', cursor: '#58a6ff' }

    const fitAddon = new FitAddon.FitAddon();

    function updateResolution() {
      const badge = document.getElementById('res-badge');
      if (badge && term) badge.innerText = `${term.cols}x${term.rows}`;
    }
    setTimeout(updateResolution, 500);

    function getTerminalBufferText(lines = 35) {
      const buffer = term.buffer.active;
      let text = [];
      const start = Math.max(0, buffer.length - lines);
      for (let i = start; i < buffer.length; i++) {
        text.push(buffer.getLine(i).translateToString(true));
      }
      return text.join('\n').trim();
    }

    window.addEventListener('resize', () => {
      fitAddon.fit();
      updateResolution();

    let socket = null;
    let wsUrl = localStorage.getItem('cs_terminal_url') || '';
    let reconnectTimeout = null;

    function getHttpBaseUrl() {
      if (!wsUrl) return '';
      return wsUrl.replace('wss://', 'https://').replace('ws://', 'http://');
    }

    function connectTerminal() {
      if (!wsUrl) return openSettings();
      if (socket) { socket.onclose = null; socket.close(); }

      term.writeln("\r\n\x1b[33mCodespace'e bağlanılıyor...\x1b[0m");
      socket = new WebSocket(wsUrl);

      socket.onopen = () => {
        clearTimeout(reconnectTimeout);
        document.getElementById('status-dot').classList.add('connected');
        term.writeln("\x1b[32m[+] Canlı Bash Terminali Hazır!\x1b[0m\r\n");
        socket.send(JSON.stringify({ type: 'resize', cols: term.cols, rows: term.rows }));
        updateResolution();
      };
      socket.onmessage = e => term.write(e.data);
      socket.onclose = () => {
        document.getElementById('status-dot').classList.remove('connected');
        term.writeln("\r\n\x1b[31m[-] Bağlantı koptu. Yeniden bağlanılıyor...\x1b[0m\r\n");
        clearTimeout(reconnectTimeout);
        reconnectTimeout = setTimeout(connectTerminal, 2500);
      };
      socket.onerror = () => document.getElementById('status-dot').classList.remove('connected');

    let aiConversationHistory = [];

    function appendMessage(sender, text, isAction = false, codeBlock = '') {
      const box = document.getElementById('ai-messages');
      const div = document.createElement('div');
      div.className = `msg ${sender}` + (isAction ? ' action' : '');
      div.innerHTML = text + (codeBlock ? `<pre>${codeBlock}</pre>` : '');
      box.appendChild(div);
      box.scrollTop = box.scrollHeight;

    async function executeCommandOnServer(command) {
      // Üstteki canlı ekrana AI'nın çalıştırdığı komutu yaz
      term.writeln(`\r\n\x1b[1;35m[AI Exec]>\x1b[0m \x1b[1;37m${command}\x1b[0m`);

      const httpBase = getHttpBaseUrl();
      const res = await fetch(`${httpBase}/api/exec`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ command })
      });
      const data = await res.json();

      // Üstteki canlı ekrana komutun çıktısını akıt
      const cleanOut = (data.stdout + '\n' + data.stderr).trim();
      if (cleanOut) {
        term.writeln(cleanOut);
      }
      return data;
    }

    async function sendAiMessage() {
      const input = document.getElementById('ai-input');
      const prompt = input.value.trim();
      if (!prompt) return;

      if (!geminiApiKey) {
        alert("Lütfen önce Ayarlar (⚙️) menüsünden Gemini API Key'inizi girin!");
        return openSettings();
      }

      input.value = '';
      appendMessage('user', prompt);
      const sendBtn = document.getElementById('btn-ai-send');
      sendBtn.disabled = true;

      const terminalScreen = getTerminalBufferText(30);

      aiConversationHistory.push({
        role: "user",
        parts: [{ text: `Kullanıcı İsteği: ${prompt}\n\nMevcut Canlı Terminal Durumu:\n\`\`\`\n${terminalScreen}\n\`\`\`` }]
      });

      try {
        let keepThinking = true;
        let loopCount = 0;

        while (keepThinking && loopCount < 5) {
          loopCount++;

          const requestBody = {
            contents: aiConversationHistory,
            systemInstruction: {
              parts: [{ text: "Sen Codespaces mobil terminalinde çalışan otonom bir Linux/AI Asistanısın. Kullanıcının isteklerini yerine getirmek için 'terminal_komut_calistir' aracını kullanarak bash komutları çalıştırabilirsin. Komut çıktılarını inceleyip net, anlaşılır Türkçe bilgi ver." }]
            },
            tools: [{
              functionDeclarations: [
                {
                  name: "terminal_komut_calistir",
                  description: "Codespaces Linux ortamında bash komutu çalıştırır.",
                  parameters: {
                    type: "OBJECT",
                    properties: {
                      komut: { type: "STRING", description: "Çalıştırılacak Linux terminal komutu" }
                    },
                    required: ["komut"]
                  }
                }
              ]
            }]
          };

          // Hem query param hem x-goog-api-key header'ı eklenerek AQ... ve AIza... anahtarları garantiye alındı
          const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${geminiModel}:generateContent?key=${geminiApiKey}`;
          const response = await fetch(apiUrl, {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
              'x-goog-api-key': geminiApiKey
            },
            body: JSON.stringify(requestBody)
          });

          const data = await response.json();
          if (data.error) throw new Error(data.error.message);

          const candidate = data.candidates?.[0]?.content;
          if (!candidate) throw new Error("Yapay zekadan yanıt alınamadı.");

          aiConversationHistory.push(candidate);
          const functionCall = candidate.parts?.find(p => p.functionCall)?.functionCall;

          if (functionCall && functionCall.name === "terminal_komut_calistir") {
            const cmdToRun = functionCall.args.komut;
            appendMessage('action', `⚡ Komut Çalıştırılıyor: <b>${cmdToRun}</b>`);
            
            const execResult = await executeCommandOnServer(cmdToRun);
            const outputText = (execResult.stdout + '\n' + execResult.stderr).trim() || '(Çıktı yok)';
            appendMessage('action', `Çıktı:`, false, outputText.slice(0, 1000));

            aiConversationHistory.push({
              role: "function",
              parts: [{
                functionResponse: {
                  name: "terminal_komut_calistir",
                  response: { output: outputText, exitCode: execResult.exitCode }
                }
              }]
            });
          } else {
            const textResponse = candidate.parts?.map(p => p.text || '').join('');
            appendMessage('ai', textResponse.replace(/\n/g, '<br>'));
            keepThinking = false;
          }
        }
      } catch (err) {
        appendMessage('ai', `<span style="color:#da3633;">❌ AI Hatası: ${err.message}</span>`);
      } finally {
        sendBtn.disabled = false;
      }
    }

    function formatSize(bytes) {
      if (bytes === 0) return '0 B';
      const k = 1024, dm = 1;
      const sizes = ['B', 'KB', 'MB', 'GB'];
      const i = Math.floor(Math.log(bytes) / Math.log(k));
      return parseFloat((bytes / Math.pow(k, i)).toFixed(dm)) + ' ' + sizes[i];
    }

    function loadFiles(dir) {
      const httpBase = getHttpBaseUrl();
      if (!httpBase) return;
      const listContainer = document.getElementById('file-list');
      listContainer.innerHTML = '<div style="padding:20px; color:#8b949e; text-align:center;">Dizin taranıyor...</div>';

      fetch(`${httpBase}/api/list?dir=${encodeURIComponent(dir)}`)
        .then(res => res.json())
        .then(data => {
          if (data.error) { listContainer.innerHTML = `<div style="padding:20px; color:#da3633;">Hata: ${data.error}</div>`; return; }
          currentDir = data.currentDir; parentDir = data.parentDir;
          document.getElementById('current-path').innerText = currentDir;
          renderFileList(data.items);
        })
        .catch(err => { listContainer.innerHTML = `<div style="padding:20px; color:#da3633;">Hata: ${err.message}</div>`; });

    function goUpDir() { if (parentDir && parentDir !== currentDir) loadFiles(parentDir); }

    function renderFileList(items) {
      const container = document.getElementById('file-list');
      container.innerHTML = '';
      if (items.length === 0) { container.innerHTML = '<div style="padding:20px; color:#8b949e; text-align:center;">Klasör boş</div>'; return; }

      items.forEach(item => {
        const row = document.createElement('div');
        row.className = 'file-item';
        row.innerHTML = `
          <div class="file-icon">${item.isDir ? '📁' : '📄'}</div>
          <div class="file-info"><div class="file-name">${item.name}</div><div class="file-meta">${item.isDir ? 'Klasör' : formatSize(item.size)}</div></div>
        `;
        let pressTimer = null, isLongPress = false, startX = 0, startY = 0;
        row.addEventListener('touchstart', (e) => {
          isLongPress = false; startX = e.touches[0].clientX; startY = e.touches[0].clientY;
          pressTimer = setTimeout(() => { isLongPress = true; if (navigator.vibrate) navigator.vibrate(40); openActionSheet(item); }, 550);
        }, { passive: true });
        row.addEventListener('touchmove', (e) => {
          if (Math.abs(e.touches[0].clientX - startX) > 10 || Math.abs(e.touches[0].clientY - startY) > 10) clearTimeout(pressTimer);
        }, { passive: true });
        row.addEventListener('touchend', () => {
          clearTimeout(pressTimer);
          if (!isLongPress && item.isDir) loadFiles(item.fullPath);
        });
        row.addEventListener('contextmenu', e => { e.preventDefault(); openActionSheet(item); });
        container.appendChild(row);
      });
    }

    function openActionSheet(item) {

    function openInReader() {

        .then(r => r.json())
        .then(data => { contentEl.innerText = data.content || data.error || 'İçerik boş.'; })
        .catch(err => { contentEl.innerText = 'Hata: ' + err.message; });

    function downloadSelectedFile() {
      if (!selectedItem || selectedItem.isDir) { alert("Klasör indirilemez."); closeSheet(); return; }
      closeSheet();
      const httpBase = getHttpBaseUrl();
      const downloadUrl = `${httpBase}/api/download?file=${encodeURIComponent(selectedItem.fullPath)}`;
      if (window.AndroidBridge && window.AndroidBridge.downloadFile) {
        window.AndroidBridge.downloadFile(downloadUrl, selectedItem.name);
      } else {
        const a = document.createElement('a'); a.href = downloadUrl; a.download = selectedItem.name; a.click();
      }
    }

      const input = document.getElementById('ai-input');
      const prompt = input.value.trim();
      if (!prompt) return;

      if (!geminiApiKey) {
        alert("Lütfen önce Ayarlar (⚙️) menüsünden Gemini API Key'inizi girin!");
        return openSettings();
      }

      input.value = '';
      appendMessage('user', prompt);
      const sendBtn = document.getElementById('btn-ai-send');
      sendBtn.disabled = true;

      const terminalScreen = getTerminalBufferText(25);

      // Kullanıcı mesajı ve canlı terminal ekranı bağlama ekleniyor
      aiConversationHistory.push({
        role: "user",
        parts: [{ text: `Kullanıcı: ${prompt}\n\n[Canlı Terminal Durumu]:\n\`\`\`\n${terminalScreen}\n\`\`\`` }]
      });

      // Bağlam çok şişmesin diye son 16 adımı tut
      if (aiConversationHistory.length > 16) {
        aiConversationHistory = aiConversationHistory.slice(-16);
      }

      try {
        let keepThinking = true;
        let loopCount = 0;

        while (keepThinking && loopCount < 5) {
          loopCount++;

          const requestBody = {
            contents: aiConversationHistory,
            systemInstruction: {
              parts: [{ text: "Sen Codespaces mobil terminalinde çalışan otonom bir Linux/AI asistanısın. Kullanıcının isteklerini yerine getirmek için 'terminal_komut_calistir' fonksiyonunu kullanarak bash komutları çalıştırabilirsin. Komut çıktısını analiz et ve iş bitince kullanıcıya net, anlaşılır Türkçe bilgi ver." }]
            },
            tools: [{
              functionDeclarations: [
                {
                  name: "terminal_komut_calistir",
                  description: "Codespaces Linux ortamında bash komutu çalıştırır.",
                  parameters: {
                    type: "OBJECT",
                    properties: {
                      komut: { type: "STRING", description: "Çalıştırılacak Linux komutu (örn: ls -la, git status)" }
                    },
                    required: ["komut"]
                  }
                }
              ]
            }]
          };

          const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${geminiModel}:generateContent?key=${geminiApiKey}`;
          const response = await fetch(apiUrl, {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
              'x-goog-api-key': geminiApiKey
            },
            body: JSON.stringify(requestBody)
          });

          const data = await response.json();
          if (data.error) throw new Error(data.error.message);

          const candidate = data.candidates?.[0]?.content;
          if (!candidate) throw new Error("Yapay zekadan yanıt alınamadı.");

          // Modelin cevabını bağlama ekle
          aiConversationHistory.push(candidate);

          const functionCall = candidate.parts?.find(p => p.functionCall)?.functionCall;

          if (functionCall && functionCall.name === "terminal_komut_calistir") {
            const cmdToRun = functionCall.args.komut;
            appendMessage('action', `⚡ Komut Koşturuluyor: <b>${cmdToRun}</b>`);

            const execResult = await executeCommandOnServer(cmdToRun);
            const outputText = (execResult.stdout + '\n' + execResult.stderr).trim() || '(Çıktı yok)';
            appendMessage('action', `Çıktı:`, false, outputText.slice(0, 800));

            // CRITICAL FIX: Google Gemini 3 kuralına göre role kesinlikle 'user' olmalıdır!
            const fnResponsePart = {
              functionResponse: {
                name: functionCall.name,
                response: { output: outputText, exitCode: execResult.exitCode }
              }
            };
            if (functionCall.id) {
              fnResponsePart.functionResponse.id = functionCall.id;
            }

            aiConversationHistory.push({
              role: "user",
              parts: [fnResponsePart]
            });

          } else {
            const textResponse = candidate.parts?.map(p => p.text || '').join('');
            appendMessage('ai', textResponse.replace(/\n/g, '<br>'));
            keepThinking = false;
          }
        }
      } catch (err) {
        appendMessage('ai', `<span style="color:#da3633;">❌ AI Hatası: ${err.message}</span>`);
      } finally {
        sendBtn.disabled = false;
      }
    }

    function formatSize(bytes) {
      if (bytes === 0) return '0 B';
      const k = 1024, dm = 1;
      const sizes = ['B', 'KB', 'MB', 'GB'];
      const i = Math.floor(Math.log(bytes) / Math.log(k));
      return parseFloat((bytes / Math.pow(k, i)).toFixed(dm)) + ' ' + sizes[i];
    }

    function loadFiles(dir) {
      const httpBase = getHttpBaseUrl();
      if (!httpBase) return;
      const listContainer = document.getElementById('file-list');
      listContainer.innerHTML = '<div style="padding:20px; color:#8b949e; text-align:center;">Dizin taranıyor...</div>';

      fetch(`${httpBase}/api/list?dir=${encodeURIComponent(dir)}`)
        .then(res => res.json())
        .then(data => {
          if (data.error) { listContainer.innerHTML = `<div style="padding:20px; color:#da3633;">Hata: ${data.error}</div>`; return; }
          currentDir = data.currentDir; parentDir = data.parentDir;
          document.getElementById('current-path').innerText = currentDir;
          renderFileList(data.items);

        .catch(err => { listContainer.innerHTML = `<div style="padding:20px; color:#da3633;">Hata: ${err.message}</div>`; });

    function renderFileList(items) {
      const container = document.getElementById('file-list');
      container.innerHTML = '';
      if (items.length === 0) { container.innerHTML = '<div style="padding:20px; color:#8b949e; text-align:center;">Klasör boş</div>'; return; }

      items.forEach(item => {
        const row = document.createElement('div');
        row.className = 'file-item';
        row.innerHTML = `
          <div class="file-icon">${item.isDir ? '📁' : '📄'}</div>
          <div class="file-info"><div class="file-name">${item.name}</div><div class="file-meta">${item.isDir ? 'Klasör' : formatSize(item.size)}</div></div>
        `;
        let pressTimer = null, isLongPress = false, startX = 0, startY = 0;
        row.addEventListener('touchstart', (e) => {
          isLongPress = false; startX = e.touches[0].clientX; startY = e.touches[0].clientY;
          pressTimer = setTimeout(() => { isLongPress = true; if (navigator.vibrate) navigator.vibrate(40); openActionSheet(item); }, 550);
        }, { passive: true });
        row.addEventListener('touchmove', (e) => {
          if (Math.abs(e.touches[0].clientX - startX) > 10 || Math.abs(e.touches[0].clientY - startY) > 10) clearTimeout(pressTimer);
        }, { passive: true });
        row.addEventListener('touchend', () => {
          clearTimeout(pressTimer);
          if (!isLongPress && item.isDir) loadFiles(item.fullPath);
        });
        row.addEventListener('contextmenu', e => { e.preventDefault(); openActionSheet(item); });
        container.appendChild(row);
      });
